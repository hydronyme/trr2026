Code ESP32 
/*
IBT-2 Motor Control Board driven by Arduino.
 
Speed and direction controlled by a potentiometer attached to analog input 0.
One side pin of the potentiometer (either one) to ground; the other side pin to +5V
 
Connection to the IBT-2 board:
IBT-2 pin 1 (RPWM) to Arduino pin 5(PWM)
IBT-2 pin 2 (LPWM) to Arduino pin 6(PWM)
IBT-2 pins 3 (R_EN), 4 (L_EN), 7 (VCC) to Arduino 5V pin
IBT-2 pin 8 (GND) to Arduino GND
IBT-2 pins 5 (R_IS) and 6 (L_IS) not connected
*/
 
int SENSOR_PIN = 0; // center pin of the potentiometer
 
int RPWM_Output = 5; // Arduino PWM output pin 5; connect to IBT-2 pin 1 (RPWM)
int LPWM_Output = 6; // Arduino PWM output pin 6; connect to IBT-2 pin 2 (LPWM)
 
void setup()
{
  pinMode(RPWM_Output, OUTPUT);
  pinMode(LPWM_Output, OUTPUT);
}
 
void loop()
{
  int sensorValue = analogRead(SENSOR_PIN);
 
  // sensor value is in the range 0 to 1023
  // the lower half of it we use for reverse rotation; the upper half for forward rotation
  if (sensorValue < 512)
  {
    // reverse rotation
    int reversePWM = -(sensorValue - 511) / 2;
    analogWrite(LPWM_Output, 0);
    analogWrite(RPWM_Output, reversePWM);
  }
  else
  {
    // forward rotation
    int forwardPWM = (sensorValue - 512) / 2;
    analogWrite(LPWM_Output, forwardPWM);
    analogWrite(RPWM_Output, 0);
  }
}

Dans le modèle du trou d'aiguille, on modélise la projection de l'espace réel vers la caméra par une droite qui passe par le trou d'aiguille et qui est aussi l'origine du repère. Son équation est définie par :

$$\left\{ \begin{array}{cl} x_v & = \ \frac{\beta_u}{\beta_v} \frac{u-u_O}{v-v_h} \\ y_v & = \ \ \beta_v \frac{1}{v-v_h} \end{array} \right. $$ $$\left\{ \begin{array}{cl} x & = \ a \times t \\ y & = \ b \times t \\ z & = \ c \times t \end{array} \right. $$

Le point de l'espace réel de coordonnées \((x_0,y_0,z_0)\) est associé par convention au paramètre \(t=1\). On a ainsi \((x_0=a,y_0=b,z_0=c)\). De plus, le point de l'espace réel se déplace sur un plan horizontal avec \(z_0=-h\). Ce point est projeté sur le plan focal de la caméra, sur le point de coordonnées \((x_1,y_1,z_1)\) avec \(x_1=-f\).

$$\left\{ \begin{array}{cl} -f & = \ x_0 \times t \\ y_1 & = \ y_0 \times t \\ z_1 & = \ -h \times t \end{array} \right. $$

Ainsi, on peut retrouver la position du point dans le monde réel à partir de la position dans la caméra :

$$\left\{ \begin{array}{cl} x_0 & = \ \frac{h \times f}{z_1} \\ y_0 & = \ - h \times \frac{y_1}{z_1} \end{array} \right. $$

/************************************************************
  ESP32 Firmware
  Servo + L298N + Encoder + PID
************************************************************/

// ================= GPIO =================
#define SERVO_PIN 18

#define MOTOR_PWM_PIN 25
#define MOTOR_IN1     26
#define MOTOR_IN2     27

#define ENC_A 34
#define ENC_B 35

// ================= PWM =================
#define SERVO_CH 0
#define MOTOR_CH 1

#define SERVO_FREQ 50
#define MOTOR_FREQ 20000

#define PWM_RES_SERVO 16
#define PWM_RES_MOTOR 10

// ================= Encoder =================
volatile long encoderCount = 0;

// ================= PID =================
float Kp = 1.2;
float Ki = 0.0;
float Kd = 0.02;

float targetPosition = 0;
float integral = 0;
float previousError = 0;

float output = 0;
const float integralLimit = 10000;

unsigned long lastPID = 0;
const int PID_INTERVAL = 5; // ms

// ================= Encoder ISR =================
void IRAM_ATTR handleEncoderA() {
  if (digitalRead(ENC_A) == digitalRead(ENC_B))
    encoderCount++;
  else
    encoderCount--;
}

// ================= Servo =================
void setServoAngle(int angle) {
  angle = constrain(angle, 0, 180);

  // 1ms -> 3277 ; 2ms -> 6553 (16 bits @ 20ms period)
  uint32_t duty = map(angle, 0, 180, 3277, 6553);
  ledcWrite(SERVO_CH, duty);
}

// ================= Motor =================
void setMotorOutput(float pwm) {
  pwm = constrain(pwm, -1023, 1023);

  if (pwm > 0) {
    digitalWrite(MOTOR_IN1, HIGH);
    digitalWrite(MOTOR_IN2, LOW);
    ledcWrite(MOTOR_CH, pwm);
  } else if (pwm < 0) {
    digitalWrite(MOTOR_IN1, LOW);
    digitalWrite(MOTOR_IN2, HIGH);
    ledcWrite(MOTOR_CH, -pwm);
  } else {
    digitalWrite(MOTOR_IN1, LOW);
    digitalWrite(MOTOR_IN2, LOW);
    ledcWrite(MOTOR_CH, 0);
  }
}

// ================= PID =================
void updatePID() {
  float position = encoderCount;
  float error = targetPosition - position;

  integral += error;
  integral = constrain(integral, -integralLimit, integralLimit);

  float derivative = error - previousError;

  output = Kp * error + Ki * integral + Kd * derivative;

  output = constrain(output, -1023, 1023);

  setMotorOutput(output);

  previousError = error;
}

// ================= Serial Parser =================
void parseCommand(String cmd) {
  cmd.trim();

  if (cmd.startsWith("S,")) {
    int angle = cmd.substring(2).toInt();
    setServoAngle(angle);
  }

  else if (cmd.startsWith("M,")) {
    targetPosition = cmd.substring(2).toFloat();
  }

  else if (cmd.startsWith("P,")) {
    int first = cmd.indexOf(',');
    int second = cmd.indexOf(',', first + 1);
    int third = cmd.indexOf(',', second + 1);

    Kp = cmd.substring(first + 1, second).toFloat();
    Ki = cmd.substring(second + 1, third).toFloat();
    Kd = cmd.substring(third + 1).toFloat();
  }

  else if (cmd == "STOP") {
    setMotorOutput(0);
  }

  else if (cmd == "R") {
    Serial.print("POS:");
    Serial.print(encoderCount);
    Serial.print(",OUT:");
    Serial.println(output);
  }
}

// ================= Setup =================
void setup() {

  Serial.begin(115200);

  pinMode(MOTOR_IN1, OUTPUT);
  pinMode(MOTOR_IN2, OUTPUT);

  pinMode(ENC_A, INPUT_PULLUP);
  pinMode(ENC_B, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(ENC_A), handleEncoderA, CHANGE);

  // Servo PWM
  ledcSetup(SERVO_CH, SERVO_FREQ, PWM_RES_SERVO);
  ledcAttachPin(SERVO_PIN, SERVO_CH);

  // Motor PWM
  ledcSetup(MOTOR_CH, MOTOR_FREQ, PWM_RES_MOTOR);
  ledcAttachPin(MOTOR_PWM_PIN, MOTOR_CH);

  lastPID = millis();
}

// ================= Loop =================
void loop() {

  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    parseCommand(cmd);
  }

  if (millis() - lastPID >= PID_INTERVAL) {
    updatePID();
    lastPID = millis();
  }
}
